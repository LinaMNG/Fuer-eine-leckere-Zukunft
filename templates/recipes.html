@app.route('/like_recipe/', methods=['POST']) # <- HIER FEHLTE @login_required def like_recipe(recipe_id): try: # Prüfen ob bereits geliked existing = db_read( "SELECT id FROM liked WHERE user_id = %s AND recipe_id = %s", (current_user.id, recipe_id) ) if existing: # Unlike db_write( "DELETE FROM liked WHERE user_id = %s AND recipe_id = %s", (current_user.id, recipe_id) ) liked = False else: # Like db_write( "INSERT INTO liked (user_id, recipe_id) VALUES (%s, %s)", (current_user.id, recipe_id) ) liked = True # Neuen Like-Count holen like_count_result = db_read( "SELECT COUNT(*) as count FROM liked WHERE recipe_id = %s", (recipe_id,) ) like_count = like_count_result[0]['count'] if like_count_result else 0 return jsonify({ 'success': True, 'liked': liked, 'like_count': like_count }) except Exception as e: print(f"Fehler beim Liken: {e}") return jsonify({'success': False, 'error': str(e)}), 500
